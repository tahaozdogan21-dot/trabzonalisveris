  if(disc>0)html+=`<div class="os-item" style="color:var(--gold)"><span>🎁 Kampanya İndirimi</span><span>-${disc} ₺</span></div>`;
  if(kartEki)html+=`<div class="os-item" style="color:#e55"><span>💳 Pos Hizmet Bedeli</span><span>+50 ₺</span></div>`;
  html+=`<div class="os-total"><span class="os-tl">Toplam (Kargo Dahil)</span><span class="os-tv">${toplam.toLocaleString('tr-TR')} ₺</span></div>`;
  document.getElementById('orderSummary').innerHTML=html;
}

function buildAddress(){
  const il=document.getElementById('chkIl').value.trim();
  const ilce=document.getElementById('chkIlce').value.trim();
  const mah=document.getElementById('chkMahalle').value.trim();
  const sokak=document.getElementById('chkSokak').value.trim();
  const tarif=document.getElementById('chkTarif').value.trim();
  return {full:`${mah}, ${sokak}, ${ilce}/${il}${tarif?' ('+tarif+')':''}`,ok:il&&ilce&&mah&&sokak};
}
